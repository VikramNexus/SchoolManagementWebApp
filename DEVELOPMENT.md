### FIFO Cash Payment Allocation Transaction

-- Lock and fetch oldest unpaid monthly fees
SELECT id, fee_month, fee_year, fee_amount, paid_amount, due_amount
FROM monthly_fees
WHERE student_id = ? AND status IN ('DUE', 'PARTIAL')
ORDER BY fee_year ASC, fee_month ASC
FOR UPDATE;
